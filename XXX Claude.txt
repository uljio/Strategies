f_array_add_pop(array, new_value_to_add) =>
    array.unshift(array, new_value_to_add)
    array.pop(array)

f_sh_sl_labels(array, swing_type) =>
    var string label_text = na
    if swing_type == 1
        label_text := array.get(array,0) >= array.get(array,1) ? 'HH' : 'LH'
        label.new(bar_index - swing_length, array.get(array,0), text = label_text, style = label.style_label_down, textcolor = swing_type_color, color = swing_type_color, size = size.tiny)
    else

f_check_overlapping(new_poi, box_array, atrValue) =>
    atr_threshold = atrValue * 2
    ok = true
    for i = 0 to array.size(box_array) - 1
        top = box.get_top(array.get(box_array, i))
        bottom = box.get_bottom(array.get(box_array, i))
        poi = (top + bottom) / 2
        upper_boundary = poi + atr_threshold
        lower_boundary = poi - atr_threshold
        if new_poi >= lower_boundary and new_poi <= upper_boundary
            ok := false
            break
    ok

f_supply_demand(value_array, bn_array, box_array, label_array, box_type, atrValue) =>
    atr_buffer = atrValue * (box_width / 10)
    box_left = array.get(bn_array, 0)
    box_right = bar_index
    var float box_top = 0.0
    var float box_bottom = 0.0
    if box_type == 1
        box_top := array.get(value_array, 0)
        box_bottom := box_top - atr_buffer
    else
        box_bottom := array.get(value_array, 0)
        box_top := box_bottom + atr_buffer
    poi = (box_top + box_bottom) / 2
    ok = f_check_overlapping(poi, box_array, atrValue)
    if ok
        box.delete(array.get(box_array, array.size(box_array) - 1))
        f_array_add_pop(box_array, box.new(left = box_left, top = box_top, right = box_right, bottom = box_bottom, border_color = box_type == 1 ? supply_outline_color : demand_outline_color, bgcolor = box_type == 1 ? supply_color : demand_color, extend = extend.right, text = box_type == 1 ? 'SUPPLY' : 'DEMAND', text_color = poi_label_color))
        box.delete(array.get(label_array, array.size(label_array) - 1))
        f_array_add_pop(label_array, box.new(left = box_left, top = poi, right = box_right, bottom = poi, border_color = poi_border_color, bgcolor = poi_border_color, extend = extend.right, text = 'POI', text_color = poi_label_color))

f_sd_to_bos(box_array, bos_array, label_array, zone_type) =>
    for i = 0 to array.size(box_array) - 1
        level_to_break = zone_type == 1 ? box.get_top(array.get(box_array, i)) : box.get_bottom(array.get(box_array, i))
        if (zone_type == 1 and close >= level_to_break) or (zone_type == -1 and close <= level_to_break)
            copied_box = box.copy(array.get(box_array, i))
            f_array_add_pop(bos_array, copied_box)
            mid = (box.get_top(array.get(box_array, i)) + box.get_bottom(array.get(box_array, i))) / 2
            box.set_top(array.get(bos_array, 0), mid)
            box.set_bottom(array.get(bos_array, 0), mid)
            box.set_extend(array.get(bos_array, 0), extend.none)
            box.set_right(array.get(bos_array, 0), bar_index)
            box.set_text(array.get(bos_array, 0), 'BOS')
            box.set_text_color(array.get(bos_array, 0), bos_label_color)
            box.delete(array.get(box_array, i))
            box.delete(array.get(label_array, i))

f_extend_box_endpoint(box_array) =>

// MA variant function
// -----------------------------------------------------------------------------
variant(type, src, len, offSig, offALMA) =>
    v1 = ta.sma(src, len)
    v2 = ta.ema(src, len)
    v3 = 2 * v2 - ta.ema(v2, len)
    v4 = 3 * (v2 - ta.ema(v2, len)) + ta.ema(ta.ema(v2, len), len)
    v5 = ta.wma(src, len)
    v6 = ta.vwma(src, len)
    v7 = 0.0
    sma1 = ta.sma(src, len)
    v7 := na(v7[1]) ? sma1 : (v7[1] * (len - 1) + src) / len
    v8 = ta.wma(2 * ta.wma(src, len / 2) - ta.wma(src, len), math.round(math.sqrt(len)))
    v9 = ta.linreg(src, len, offSig)
    v10 = ta.alma(src, len, offALMA, offSig)
    v11 = ta.sma(v1, len)
    a1 = math.exp(-1.414 * 3.14159 / len)
    b1 = 2 * a1 * math.cos(1.414 * 3.14159 / len)
    c2 = b1
    c3 = -a1 * a1
    c1 = 1 - c2 - c3
    v12 = 0.0
    v12 := c1 * (src + nz(src[1])) / 2 + c2 * nz(v12[1]) + c3 * nz(v12[2])
    type == 'EMA' ? v2 : type == 'DEMA' ? v3 : type == 'TEMA' ? v4 : type == 'WMA' ? v5 : type == 'VWMA' ? v6 : type == 'SMMA' ? v7 : type == 'HullMA' ? v8 : type == 'LSMA' ? v9 : type == 'ALMA' ? v10 : type == 'TMA' ? v11 : type == 'SSMA' ? v12 : v1

var line tp2Line = na
var line tp3Line = na

var label entryLabel = na
var label slLabel = na
var label tp1Label = na
var label tp2Label = na
var label tp3Label = na

// Create new lines when position opens
if positionJustOpened and not na(actualEntryPrice)
    // Delete old lines
    line.delete(entryLine)
    line.delete(slLine)
    line.delete(tp1Line)
    line.delete(tp2Line)
    line.delete(tp3Line)
    label.delete(entryLabel)
    label.delete(slLabel)
    label.delete(tp1Label)
    label.delete(tp2Label)
    label.delete(tp3Label)
    
    // Create new lines anchored to entry bar
    entryLine := line.new(bar_index, actualEntryPrice, bar_index + 50, actualEntryPrice, color = color.blue, width = 2, style = line.style_solid)
    slLine := line.new(bar_index, slLevel, bar_index + 50, slLevel, color = color.red, width = 2, style = line.style_solid)
    tp1Line := line.new(bar_index, tp1Level, bar_index + 50, tp1Level, color = color.green, width = 1, style = line.style_solid)
    tp2Line := line.new(bar_index, tp2Level, bar_index + 50, tp2Level, color = color.new(color.green, 30), width = 1, style = line.style_solid)
    tp3Line := line.new(bar_index, tp3Level, bar_index + 50, tp3Level, color = color.new(color.green, 60), width = 1, style = line.style_solid)
    
    // Create labels
    entryLabel := label.new(bar_index, actualEntryPrice, "Entry: " + str.tostring(actualEntryPrice, format.mintick), color = color.blue, textcolor = color.white, style = label.style_label_left, size = size.small)
    slLabel := label.new(bar_index, slLevel, "SL: " + str.tostring(slLevel, format.mintick), color = color.red, textcolor = color.white, style = label.style_label_left, size = size.small)
    tp1Label := label.new(bar_index, tp1Level, "TP1: " + str.tostring(tp1Level, format.mintick), color = color.green, textcolor = color.white, style = label.style_label_left, size = size.small)

// Extend lines on each bar while position is open
if strategy.position_size != 0 and not na(entryLine)
    line.set_x2(entryLine, bar_index + 50)
    line.set_x2(slLine, bar_index + 50)
    line.set_x2(tp1Line, bar_index + 50)
    line.set_x2(tp2Line, bar_index + 50)
    line.set_x2(tp3Line, bar_index + 50)

// Delete lines when position closes
if positionJustClosed
    line.delete(entryLine)
    line.delete(slLine)
    line.delete(tp1Line)
    line.delete(tp2Line)
    line.delete(tp3Line)
    label.delete(entryLabel)
    label.delete(slLabel)
    label.delete(tp1Label)
    label.delete(tp2Label)
    label.delete(tp3Label)
    entryLine := na
    slLine := na
    tp1Line := na
    tp2Line := na
    tp3Line := na

plotshape(i_alertOn and shortFilled, 'Short Entry', shape.labeldown, location.abovebar, color.new(color.red, 0), text = 'SHORT', textcolor = color.white, size = size.normal)
plotshape(i_alertOn and lxTrigger and strategy.position_size > 0, 'Long Exit Signal', shape.xcross, location.abovebar, color.new(color.orange, 0), text = 'EXIT', textcolor = color.white, size = size.small)
plotshape(i_alertOn and sxTrigger and strategy.position_size < 0, 'Short Exit Signal', shape.xcross, location.belowbar, color.new(color.orange, 0), text = 'EXIT', textcolor = color.white, size = size.small)

barcolor(i_barColOn and strategy.position_size > 0 ? color.new(color.green, 70) : i_barColOn and strategy.position_size < 0 ? color.new(color.red, 70) : na)

    table.cell(debugTable, 1, 10, (inBullSignal ? "B" : "") + (inBearSignal ? "S" : "") + (not inBullSignal and not inBearSignal ? "NONE" : ""),
